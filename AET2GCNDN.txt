local function playTeleportSound()
    if not States.UltraInstinctSound then return end
    local soundId = teleportSoundIDs[math.random(1, #teleportSoundIDs)]
    local sound = Instance.new("Sound")
    sound.SoundId = soundId
    sound.Volume = 0.7
    sound.Parent = Workspace
    sound.Ended:Connect(function() pcall(function() sound:Destroy() end) end)
    pcall(function() sound:Play() end)
end

local function isPositionClear(fromPos, toPos, ignoreList)
    local dir = toPos - fromPos
    local dist = dir.Magnitude
    if dist <= 0.01 then return true end
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = ignoreList or {}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local result = Workspace:Raycast(fromPos, dir.Unit * dist, params)
    return result == nil
end

local function findSafeTeleport(myPos, intrPos, baseOffset, ignoreList)
    local dir = myPos - intrPos
    dir = Vector3.new(dir.X, 0, dir.Z)
    if dir.Magnitude < 0.001 then dir = Vector3.new(1,0,0) else dir = dir.Unit end
    local left = Vector3.new(-dir.Z, 0, dir.X)
    local right = Vector3.new(dir.Z, 0, -dir.X)
    local candidates = {myPos + left * baseOffset, myPos + right * baseOffset}
    for _, cand in ipairs(candidates) do
        local fromPos = myPos + Vector3.new(0, 2, 0)
        local toPos = Vector3.new(cand.X, myPos.Y, cand.Z) + Vector3.new(0, 2, 0)
        if isPositionClear(fromPos, toPos, ignoreList) then return toPos end
    end
    return nil
end

local function clearUltraInstinctRing()
    if ringUpdateConn then ringUpdateConn:Disconnect(); ringUpdateConn = nil end
    if ringModel and ringModel.Parent then ringModel:Destroy(); ringModel = nil end
    segmentParts = {}
end

local function createUltraInstinctRing()
    clearUltraInstinctRing()
    local radius = States.UltraInstinctRange
    local char = LocalPlayer.Character
    if not char or not char.PrimaryPart then return end
    local root = char.PrimaryPart
    ringModel = Instance.new("Model")
    ringModel.Name = "UltraInstinctRing"
    ringModel.Parent = Workspace
    local segments = 36
    for i = 1, segments do
        local angle = (i-1) * (2 * math.pi / segments)
        local seg = Instance.new("Part")
        seg.Anchored = true; seg.CanCollide = false; seg.CastShadow = false
        seg.Size = Vector3.new(1.0, 0.06, 0.5)
        seg.Material = Enum.Material.Neon
        seg.Color = Color3.fromRGB(240, 30, 30)
        seg.Parent = ringModel
        local pos = Vector3.new(math.cos(angle) * radius, 0.12, math.sin(angle) * radius)
        seg.CFrame = CFrame.new(root.Position + pos) * CFrame.Angles(0, -angle, 0)
        table.insert(segmentParts, seg)
    end
    
    ringUpdateConn = RunService.RenderStepped:Connect(function()
        if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then clearUltraInstinctRing(); return end
        local centerPos = LocalPlayer.Character.PrimaryPart.Position
        for i, seg in ipairs(segmentParts) do
            if seg and seg.Parent then
                local angle = (i-1) * (2 * math.pi / #segmentParts)
                local x = math.cos(angle) * States.UltraInstinctRange
                local z = math.sin(angle) * States.UltraInstinctRange
                seg.CFrame = CFrame.new(centerPos + Vector3.new(x, 0.12, z)) * CFrame.Angles(0, -angle, 0)
            end
        end
        for _, pl in pairs(Players:GetPlayers()) do
            if pl ~= LocalPlayer then
                local ch = pl.Character
                if ch and ch.PrimaryPart and ch:FindFirstChildWhichIsA("Humanoid") then
                    local dist = (ch.PrimaryPart.Position - centerPos).Magnitude
                    if dist <= States.UltraInstinctRange then
                        local now = tick()
                        if now - _lastTeleportTime >= 0.001 then
                            local safe = findSafeTeleport(centerPos, ch.PrimaryPart.Position, States.UltraInstinctRange + 3, {LocalPlayer.Character, ringModel})
                            if safe then
                                local primary = LocalPlayer.Character and LocalPlayer.Character.PrimaryPart
                                if primary then
                                    primary.CFrame = CFrame.new(safe + Vector3.new(0, 0.5, 0))
                                    playTeleportSound()
                                    _lastTeleportTime = now
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
end

-- ESP System Backend
local ESPFolder = Instance.new("Folder")

local ESPCache = {}

local function RemoveESP(t) 
    if ESPCache[t] then 
        if ESPCache[t].Bill then ESPCache[t].Bill:Destroy() end 
        if ESPCache[t].Highlight then ESPCache[t].Highlight:Destroy() end 
        if ESPCache[t].Box then ESPCache[t].Box:Destroy() end 
        ESPCache[t] = nil 
    end 
end

local function GetOrCreatePlayerESP(character, playerName)
    if ESPCache[character] then return ESPCache[character] end
    local head = character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    if not head then return end
    local espCol = Color3.fromRGB(74, 222, 128)
    local bill = Instance.new("BillboardGui")
    bill.AlwaysOnTop = true; bill.Size = UDim2.new(0, 160, 0, 40); bill.StudsOffset = Vector3.new(0, 3, 0); bill.Adornee = head; bill.Parent = ESPFolder
    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1, 0, 1, 0); txt.BackgroundTransparency = 1; txt.Text = "[ผู้เล่น] " .. playerName; txt.TextColor3 = espCol; txt.TextSize = 13; txt.Font = Enum.Font.GothamBold; txt.Parent = bill
    local highlight = Instance.new("Highlight")
    highlight.Adornee = character; highlight.FillColor = espCol; highlight.FillTransparency = 0.65; highlight.OutlineColor = espCol; highlight.Parent = character
    ESPCache[character] = { Bill = bill, Highlight = highlight }
    return ESPCache[character]
end

local function GetOrCreateGenericESP(tgt, color, label)
    if ESPCache[tgt] then return ESPCache[tgt] end
    local p = tgt:IsA("Model") and (tgt:FindFirstChild("HumanoidRootPart") or tgt.PrimaryPart) or tgt; if not p then return end
    local bill = Instance.new("BillboardGui"); bill.AlwaysOnTop = true; bill.Size = UDim2.new(0, 140, 0, 40); bill.StudsOffset = Vector3.new(0, 2.5, 0); bill.Adornee = p; bill.Parent = ESPFolder
    local txt = Instance.new("TextLabel"); txt.Size = UDim2.new(1, 0, 1, 0); txt.BackgroundTransparency = 1; txt.Text = label; txt.TextColor3 = color; txt.TextSize = 12; txt.Font = Enum.Font.GothamBold; txt.Parent = bill
    local box = Instance.new("BoxHandleAdornment"); box.AlwaysOnTop = true; box.ZIndex = 5; box.Size = tgt:IsA("Model") and tgt:GetExtentsSize() or Vector3.new(3, 3, 3); box.Color3 = color; box.Transparency = 0.5; box.Adornee = p; box.Parent = p
    ESPCache[tgt] = { Bill = bill, Box = box }
    return ESPCache[tgt]
end

-- Reset Hitbox Backend Function
local function ResetHitboxes()
    for player, data in pairs(OriginalHitboxData) do
        if player and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = data.Size or Vector3.new(2, 2, 1)
                hrp.Transparency = data.Transparency or 1
                hrp.Material = data.Material or Enum.Material.Plastic
            end
        end
    end
    OriginalHitboxData = {}
end

-- Config Manager Functions with List Support
local function GetSavedConfigs()
    local list = {}
    if listfiles and isfolder(CONFIG_FOLDER) then
        for _, file in ipairs(listfiles(CONFIG_FOLDER)) do
            local name = file:match(".+/(.+)%.json$") or file:match(".+\\(.+)%.json$")
            if name then table.insert(list, name) end
        end
    end
    if #list == 0 then table.insert(list, "ไม่มีไฟล์เซฟ") end
    return list
end

local function SaveConfig()
    if writefile then
        if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end
        pcall(function() writefile(CONFIG_FOLDER .. "/" .. CurrentConfigName .. ".json", HttpService:JSONEncode(States)) end)
        WindUI:Notify({ Title = "ระบบบันทึก Config", Content = "บันทึกข้อมูล '" .. CurrentConfigName .. "' เรียบร้อย!", Duration = 3 })
    end
end

local function LoadConfig()
    if readfile and isfile(CONFIG_FOLDER .. "/" .. CurrentConfigName .. ".json") then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FOLDER .. "/" .. CurrentConfigName .. ".json")) end)
        if success and type(data) == "table" then
            for k, v in pairs(data) do States[k] = v end
            WindUI:Notify({ Title = "ระบบบันทึก Config", Content = "โหลด Config '" .. CurrentConfigName .. "' สำเร็จ!", Duration = 3 })
        end
    else
        WindUI:Notify({ Title = "ระบบบันทึก Config", Content = "ไม่พบไฟล์ Config นี้!", Duration = 3 })
    end
end

local function DeleteConfig()
    if delfile and isfile(CONFIG_FOLDER .. "/" .. CurrentConfigName .. ".json") then
        pcall(function() delfile(CONFIG_FOLDER .. "/" .. CurrentConfigName .. ".json") end)
        WindUI:Notify({ Title = "ระบบจัดการ Config", Content = "ลบโปรไฟล์ '" .. CurrentConfigName .. "' เรียบร้อยแล้ว", Duration = 3 })
    end
end

LocalPlayer.CharacterAdded:Connect(function(newChar)
    if States.LowFriction then
        task.spawn(function()
            task.wait(0.5)
            for _, part in pairs(newChar:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CustomPhysicalProperties = PhysicalProperties.new(0.01, 0, 0, 0, 0)
                end
            end
        end)
    end
end)

-- Rhythm Walk Loop
task.spawn(function()
    while true do
        task.wait(0.1)
        if States.RhythmWalk then
            local hum = SafeGetHumanoid()
            if hum then
                hum.WalkSpeed = States.RhythmSpeed
                for i = 1, math.floor(States.RhythmInterval * 10) do
                    if not States.RhythmWalk then break end
                    task.wait(0.1)
                end
                
                if States.RhythmWalk then
                    hum.WalkSpeed = 16
                    for i = 1, math.floor(States.RhythmInterval * 10) do
                        if not States.RhythmWalk then break end
                        task.wait(0.1)
                    end
                else
                    hum.WalkSpeed = 16
                end
            end
        end
    end
end)

-- Waypoint Fly Loop
task.spawn(function()
    while true do
        task.wait(0.1)
        if States.SavedPointsFly and #SavedWaypoints > 0 and not isFlyingToPoints then
            isFlyingToPoints = true
            for _, targetCF in ipairs(SavedWaypoints) do
                if not States.SavedPointsFly then break end
                local hrp = SafeGetHRP()
                if hrp then
                    local distance = (hrp.Position - targetCF.Position).Magnitude
                    while distance > 5 and States.SavedPointsFly do
                        hrp.CFrame = hrp.CFrame:Lerp(targetCF, math.clamp(States.PointFlySpeed / (distance * 10), 0.05, 1))
                        distance = (hrp.Position - targetCF.Position).Magnitude
                        task.wait()
                    end
                end
                task.wait(0.5)
            end
            isFlyingToPoints = false
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if States.Speed and not States.RhythmWalk then
        local hum = SafeGetHumanoid()
        if hum then hum.WalkSpeed = States.SpeedValue end
    end

    if States.Noclip then
        local c = SafeGetCharacter()
        if c then
            for _, part in pairs(c:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end

    if States.Hitbox then
        pcall(function()
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        if not OriginalHitboxData[player] then
                            OriginalHitboxData[player] = {
                                Size = hrp.Size,
                                Transparency = hrp.Transparency,
                                Material = hrp.Material
                            }
                        end
                        local s = States.HitboxValue
                        hrp.Size = Vector3.new(s, s, s)
                        hrp.Transparency = States.HitboxTrans
                        hrp.Color = Color3.fromRGB(255, 0, 0)
                        hrp.Material = Enum.Material.Neon
                        hrp.CanCollide = false
                    end
                end
            end
        end)
    end

    if States.UltraInstinct and not ringModel then createUltraInstinctRing() end

    -- Circle Lock Visualizer Logic
    if States.CircleLock then
        CircleDrawing.Visible = true
        CircleDrawing.Radius = States.CircleLockRadius
        CircleDrawing.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

        local closestPlayer = nil
        local shortestDist = States.CircleLockRadius
        local screenCenter = CircleDrawing.Position
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local head = p.Character:FindFirstChild("Head") or p.Character:FindFirstChild("HumanoidRootPart")
                if head then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                        if dist <= States.CircleLockRadius and dist < shortestDist then
                            shortestDist = dist; closestPlayer = p
                        end
                    end
                end
            end
        end
        if closestPlayer and closestPlayer.Character then
            local targetHead = closestPlayer.Character:FindFirstChild("Head") or closestPlayer.Character:FindFirstChild("HumanoidRootPart")
            if targetHead then Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetHead.Position) end
        end
    else
        CircleDrawing.Visible = false
    end

    if States.CustomFOV then Camera.FieldOfView = States.FOV else Camera.FieldOfView = 70 end
    if States.CustomZoom then LocalPlayer.CameraMaxZoomDistance = 10000 else LocalPlayer.CameraMaxZoomDistance = 400 end

    if States.AntiRagdoll then
        local humRef = SafeGetHumanoid()
        if humRef then
            local state = humRef:GetState()
            if humRef.PlatformStand or state == Enum.HumanoidStateType.Physics or state == Enum.HumanoidStateType.Ragdoll or state == Enum.HumanoidStateType.FallingDown or state == Enum.HumanoidStateType.GettingUp then
                humRef.PlatformStand = false
                humRef:ChangeState(Enum.HumanoidStateType.Running)
            end
        end
    end

    if States.AntiFling then
        local hrp = SafeGetHRP()
        if hrp then
            if hrp.AssemblyLinearVelocity.Magnitude > 150 or hrp.AssemblyAngularVelocity.Magnitude > 100 then
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end
        end
    end

    if States.InstantInteract then
        pcall(function()
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("ProximityPrompt") then
                    obj.HoldDuration = 0
                    obj.RequiresLineOfSight = false
                    obj.MaxActivationDistance = math.max(obj.MaxActivationDistance, 30)
                    fireproximityprompt(obj)
                end
            end
        end)
    end

    if States.Fullbright then
        Lighting.Brightness = 2; Lighting.ClockTime = 14; Lighting.GlobalShadows = false; Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
    end

    if States.GodMode then 
        local humRef = SafeGetHumanoid()
        if humRef then humRef.MaxHealth = math.huge; humRef.Health = math.huge end 
    end

    if States.WaterWalk then 
        local hrp = SafeGetHRP() 
        local humRef = SafeGetHumanoid()
        if hrp and humRef then 
            local ray = Ray.new(hrp.Position, Vector3.new(0, -8, 0)) 
            local hit = Workspace:FindPartOnRay(ray, SafeGetCharacter()) 
            if hit and (hit.Name:lower():find("water") or hit.Material == Enum.Material.Water) then 
                hrp.Velocity = Vector3.new(hrp.Velocity.X, 0, hrp.Velocity.Z) 
                if humRef:GetState() == Enum.HumanoidStateType.Freefall then humRef:ChangeState(Enum.HumanoidStateType.Running) end 
            end 
        end 
    end

    if States.FollowTarget and States.SelectedTargetPlayer and States.SelectedTargetPlayer.Character then
        local targetHRP = States.SelectedTargetPlayer.Character:FindFirstChild("HumanoidRootPart")
        local myHRP = SafeGetHRP()
        if targetHRP and myHRP then
            local offsetPos = targetHRP.Position - targetHRP.CFrame.LookVector * States.TPDepth
            myHRP.CFrame = CFrame.new(offsetPos, targetHRP.Position)
        end
    end

    if States.TPBelowFeet and States.SelectedTargetPlayer and States.SelectedTargetPlayer.Character then
        local targetHRP = States.SelectedTargetPlayer.Character:FindFirstChild("HumanoidRootPart")
        local myHRP = SafeGetHRP()
        if targetHRP and myHRP then
            local belowPos = targetHRP.Position + Vector3.new(0, -States.TPDepth, 0)
            myHRP.CFrame = CFrame.new(belowPos, targetHRP.Position + Vector3.new(0, 2, 0))
        end
    end
end)
